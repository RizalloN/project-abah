php artisan tinker

use App\Support\DashboardSmeSegmentService;
use App\Support\RkaLookupService;
use Carbon\Carbon;
use Illuminate\Support\Facades\DB;

echo "\n=== Direct RKA Loading Test ===\n\n";

// Test the RkaLookupService directly
$rkaService = app(RkaLookupService::class);

// Get definitions for SME OS
$definitions = [
    'kecil_non_cashcoll_os' => ['mata_anggaran' => ['B.2.a. Kredit Kecil Non Cash Collateral'], 'uker_contains_any' => ['KC', 'KCP']],
    'cashcoll_os' => ['mata_anggaran' => ['B.2.b. Kredit Kecil Cash Collateral'], 'uker_contains_any' => ['KC', 'KCP']],
];

echo "1. Testing regional filter with 'DPK' mata_anggaran (for SML):\n";

$smlDefinitions = [
    'kecil_non_cashcoll_sml' => ['mata_anggaran' => ['DPK Rp Kecil Non Cash Collateral'], 'uker_contains_any' => ['KC', 'KCP']],
];

// Try to load regional data
$regionFilters = ['MADIUN', 'NGAWI', 'MAGETAN'];
$result = $rkaService->aggregateByGroupWithRegionalFilter($smlDefinitions, 'apr', $regionFilters, 2026);

echo "   Result:\n";
foreach ($result as $defKey => $regions) {
    foreach ($regions as $region => $value) {
        echo "   - $defKey | $region: " . number_format($value, 0) . "\n";
    }
}

echo "\n2. Checking raw RKA data for 'DPK Rp Kecil Non Cash Collateral':\n";
$count = DB::table('rka')
    ->where('mata_anggaran', 'DPK Rp Kecil Non Cash Collateral')
    ->count();
echo "   Total rows with this mata_anggaran: $count\n";

// Check for regional pattern
$sampleRows = DB::table('rka')
    ->where('mata_anggaran', 'DPK Rp Kecil Non Cash Collateral')
    ->whereIn('kanca', ['KC Ponorogo'])
    ->whereRaw("(desc_uker LIKE '%MADIUN%' OR desc_uker LIKE '%NGAWI%' OR desc_uker LIKE '%MAGETAN%')")
    ->limit(5)
    ->get(['kanca', 'desc_uker', 'mata_anggaran', 'apr']);

if ($sampleRows->count() > 0) {
    echo "   Sample rows:\n";
    foreach ($sampleRows as $row) {
        echo "   - desc_uker: " . $row->desc_uker . " | apr: " . number_format($row->apr, 0) . "\n";
    }
} else {
    echo "   No regional data found for this mata_anggaran!\n";
}

exit();
