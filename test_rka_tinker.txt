php artisan tinker

use App\Support\DashboardSmeSegmentService;
use Carbon\Carbon;

// Create service instance
$service = app(DashboardSmeSegmentService::class);
$period = '2026-04-20';

// Directly test the RKA loading
$rka = $service->loadRkaForSegment($period, 'SME', ['KC Ponorogo', 'KC Madiun', 'KC Ngawi', 'KC Magetan']);

// Check what we got
echo "\n=== RKA DATA TEST ===\n";
echo "Period: $period\n";
echo "Requested branches: KC Ponorogo, KC Madiun, KC Ngawi, KC Magetan\n\n";

// Summarize RKA data by branch
$totals = [];
foreach ($rka as $type => $typeData) {
    foreach ($typeData as $category => $categoryData) {
        foreach ($categoryData as $branch => $values) {
            if (!isset($totals[$branch])) {
                $totals[$branch] = ['m1' => 0, 'current' => 0];
            }
            $totals[$branch]['m1'] += $values['m1'] ?? 0;
            $totals[$branch]['current'] += $values['current'] ?? 0;
        }
    }
}

echo "RKA Totals by Branch:\n";
foreach (['KC PONOROGO', 'KC MADIUN', 'KC NGAWI', 'KC MAGETAN'] as $branch) {
    $data = $totals[$branch] ?? null;
    if ($data) {
        $total = $data['m1'] + $data['current'];
        if ($total > 0) {
            echo "  ✓ $branch: Rp " . number_format($total, 0) . " (M-1: " . number_format($data['m1'], 0) . ", Current: " . number_format($data['current'], 0) . ")\n";
        } else {
            echo "  ✗ $branch: Rp 0 (NO DATA)\n";
        }
    } else {
        echo "  ✗ $branch: NOT FOUND\n";
    }
}

exit();
